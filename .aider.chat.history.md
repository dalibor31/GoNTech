// BruteforceRepo handles brute force protection tracking
type BruteforceRepo struct {
	db *sql.DB
}

// NoviBruteforceRepo kreira novi BruteforceRepo
func NoviBruteforceRepo(db *sql.DB) *BruteforceRepo {
	return &BruteforceRepo{db: db}
}

// ProveriIliJeZablokovan proverava da li je korisnik ili IP blokiran
func (r *BruteforceRepo) ProveriIliJeZablokovan(ctx context.Context, korisnickoIme string, ipAdresa string) (*model.BruteforceStatus, error) {
	var status model.BruteforceStatus
	var blockedUntil time.Time
	
	err := r.db.QueryRowContext(ctx, `
		SELECT 
			korisnicko_ime, 
			blokiran, 
			blokirano_do, 
			pokusaji, 
			ip_adresa, 
			najnoviji_pokusaj
		FROM bruteforce WHERE korisnicko_ime = ? OR ip_adresa = ?`,
		korisnickoIme, ipAdresa).Scan(
		&status.KorisnickoIme, &status.Blokiran, &blockedUntil, &status.Pokusaji, &status.IPAdresa, &status.NajnovijiPokusaj)
	
	if err == sql.ErrNoRows {
		return nil, nil // korisnik nije u tabeli, nema problema
	}
	
	if err != nil {
		return nil, fmt.Errorf("ntech: BruteforceRepo.ProveriIliJeZablokovan: %w", err)
	}
	
	status.BlokiranoDo = blockedUntil
	
	return &status, nil
}

// ZabiljeziPokusaj zabilježava novi pokušaj prijave (uspešan ili neuspešan)
func (r *BruteforceRepo) ZabiljeziPokusaj(ctx context.Context, korisnickoIme string, ipAdresa string, uspesno bool) error {
	now := time.Now()
	
	// Ako je pokušaj uspešan, resetuj sve brojače
	if uspesno {
		_, err := r.db.ExecContext(ctx, `
			DELETE FROM bruteforce WHERE korisnicko_ime = ? OR ip_adresa = ?`,
			korisnickoIme, ipAdresa)
		return err
	}
	
	// Ako je neuspešan, zabilježi pokušaj
	var trenutniPokusaji int
	
	err := r.db.QueryRowContext(ctx, `
		SELECT COALESCE(MAX(pokusaji), 0) as trenutni_pokusaji 
		FROM bruteforce WHERE korisnicko_ime = ? OR ip_adresa = ?`,
		korisnickoIme, ipAdresa).Scan(&trenutniPokusaji)
	
	if err != nil {
		return fmt.Errorf("ntech: BruteforceRepo.ZabiljeziPokusaj: %w", err)
	}
	
	// Ako je već blokiran, samo ažuriraj datum do kada je blokiran
	var trenutnoBlokiran bool
	var trenutnoBlokiranoDo time.Time
	
	r.db.QueryRowContext(ctx, `
		SELECT blokirano, blokirano_do FROM bruteforce WHERE korisnicko_ime = ? OR ip_adresa = ?`,
		korisnickoIme, ipAdresa).Scan(&trenutnoBlokiran, &trenutnoBlokiranoDo)
	
	if trenutnoBlokiran {
		// Ako je već blokiran, samo ažuriraj datum do kada je blokiran ako je proteklo vreme
		if now.After(trenutnoBlokiranoDo) {
			_, err = r.db.ExecContext(ctx, `
				UPDATE bruteforce SET 
					blokiran = 1, 
					blokirano_do = ?, 
					pokusaji = ? + ?, 
					najnoviji_pokusaj = ?
				WHERE korisnicko_ime = ? OR ip_adresa = ?`,
				now.Add(5*time.Minute), trenutniPokusaji+1, 1, now, korisnickoIme, ipAdresa)
			return err
		}
		return nil // Još uvek blokiran
	}
	
	// Novi pokušaj - ažuriraj broj pokušaja
	_, err = r.db.ExecContext(ctx, `
		INSERT INTO bruteforce (korisnicko_ime, ip_adresa, blokirano, blokirano_do, 
			pokusaji, najnoviji_pokusaj)
		VALUES (?, ?, 0, NULL, ?, ?)
		ON CONFLICT(korisnicko_ime, ip_adresa) DO UPDATE SET
			pokusaji = bruteforce.pokusaji + excluded.pokusaji,
			najnoviji_pokusaj = excluded.najnoviji_pokusaj`,
		korisnickoIme, ipAdresa, trenutniPokusaji+1, now)
	
	if err != nil {
		return fmt.Errorf("ntech: BruteforceRepo.ZabiljeziPokusaj: %w", err)
	}
	
	// Ako je dostignut prag pokušaja (5), blokiraj korisnika/IP na 5 minuta
	if trenutniPokusaji+1 >= 5 {
		_, err = r.db.ExecContext(ctx, `
			UPDATE bruteforce SET 
				blokiran = 1, 
				blokirano_do = datetime('now', '+5 minutes'),
				pokusaji = pokusaji + 1
			WHERE korisnicko_ime = ? OR ip_adresa = ?`,
			korisnickoIme, ipAdresa)
		if err != nil {
			return fmt.Errorf("ntech: BruteforceRepo.ZabiljeziPokusaj: %w", err)
		}
	}
	
	return nil
}

// getClientIP vraća IP adresu klijenta iz zahteva
func getClientIP(r *http.Request) string {
	forwarded := r.Header.Get("X-Forwarded-For")
	if forwarded != "" {
		return strings.Split(forwarded, ",")[0]
	}
	ip := r.RemoteAddr
	if idx := strings.LastIndex(ip, ":"); idx != -1 {
		ip = ip[:idx]
	}
	return ip
}
